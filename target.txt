        }

        // Öneri: Rastgelelik ile Eşit Skorları Kırmak (Random Tie-Breaking)
        // Eğer birden fazla boş yer varsa (hepsinin skoru 0), restartCount > 0 olduğunda
        // aralarına küçük bir rastgele sayı ekliyoruz ki her yeniden başlatmada farklı bir boşluğu denesin.
        const noise = restartCount > 0 ? random() : 0;

        return { ...cand, score: score + failurePenalty + noise };
      });

      candidatesWithScores.sort((a, b) => a.score - b.score);
